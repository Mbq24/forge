DSL: vt-breakout.yaml

# VT-Breakout — EMA alignment + candlestick confirmation
name: "VT-Breakout"
timeframe: "1h"
indicators:
  - id: ema_5    type: ema    params: { period: 5 }
  - id: ema_8    type: ema    params: { period: 8 }
  - id: ema_13   type: ema    params: { period: 13 }
  - id: ema_21   type: ema    params: { period: 21 }
  - id: rsi     type: rsi    params: { period: 14 }
  - id: atr     type: atr    params: { period: 14 }
compounds:
  - id: alignment  type: ema_alignment
    params: { emas: [ema_5, ema_8, ema_13, ema_21] }
  - id: spread    type: ema_spread
    params: { emas: [ema_5, ema_8, ema_13, ema_21] }
  - id: pull      type: pull_count
    params: { ema: ema_5 }
patterns: [hammer, bullish_engulfing, doji]
signals:
  entry: "alignment == 1 AND (hammer OR bullish_engulfing)"
  exit:  "pull >= 3 OR rsi > 70"

Generated Pine Script v5

//@version=5
indicator("VT-Breakout", overlay=false)

// Standard indicators
ema_5  = ta.ema(close, 5)
ema_8  = ta.ema(close, 8)
ema_13 = ta.ema(close, 13)
ema_21 = ta.ema(close, 21)
rsi    = ta.rsi(close, 14)
atr    = ta.atr(14)

// VT concepts
alignment = ema_5 > ema_8 and ema_8 > ema_13
         and ema_13 > ema_21 ? 1 : (
         ema_5 < ema_8 and ema_8 < ema_13
         and ema_13 < ema_21 ? -1 : 0)

spread = (math.max(ema_5, ema_8, ema_13, ema_21)
       - math.min(ema_5, ema_8, ema_13, ema_21)) / close

var pull = 0
pull := close > ema_5 ? nz(pull[1]) + 1 : 0

hammer = ta.hammer(open, high, low, close)
bullish_engulfing = ta.bullishengulfing(open, high, low, close)

// Signals
entry = (alignment == 1.0 and
         (hammer or bullish_engulfing))
exit  = (pull >= 3.0 or rsi > 70.0)

plot(ema_5, "EMA 5", color.blue)
plot(ema_8, "EMA 8", color.green)
plotshape(entry, style=shape.triangleup,
          location=location.belowbar, color=color.green)
plotshape(exit, style=shape.triangledown,
          location=location.abovebar, color=color.red)

VT Concepts (from volatility-trader Master Indicator Suite)

compound EMA Alignment
Bullish ordering check across a family of EMAs (fastest → slowest). Returns +1 (full bull), -1 (full bear), or 0 (mixed).
alignment = bull_ordered ? 1 : bear_ordered ? -1 : 0
compound EMA Spread
Distance between widest and narrowest EMA, normalised by close. Tight spread = consolidation = breakout imminent.
spread = (max_ema - min_ema) / close
compound Candle Proximity
Distance from close to a reference EMA, expressed in ATR units. Body touching EMA = entry trigger (Legacy Class concept).
proximity = (close - ema) / atr(14)
compound Pull Count
Consecutive closes above the 5 EMA. After 3+, expect a pullback / mean reversion candle.
pull = close > ema_5 ? prev + 1 : 0
pattern Candlestick Patterns
10 built-in patterns: doji, hammer, shooting_star, bullish/bearish_engulfing, harami, morning/evening_star, three_white_soldiers, three_black_crows.
ta.hammer(open, high, low, close)
feature Session Encoding
Asian (0-8 UTC), London (8-16), NY (13-22), London+NY overlap. Used in signal conditions for time-aware trading.
session_ny = hour >= 13 and hour < 22
